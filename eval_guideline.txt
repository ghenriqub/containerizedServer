## BASIC

-- docker commands
docker ps
docker ps -a
docker run -it <name> sh/bash
docker run -it -v <volName>:<volPath> <containerName> sh/bash
docker system prune -a --volumes
docker volume ls
docker volume rm <name>

-- eval commands
docker exec <container> cat /etc/os-release
docker compose -f srcs/docker-compose.yml build mariadb

-- network
docker network inspect inception_inception -f '{{range .Containers}}{{.Name}} {{end}}'

-- test certificate
openssl s_client -connect ghenriqu.42.fr:443 -tls1_2 </dev/null 2>&1 | grep -E "Protocol|Cipher"

-- testar tls
curl -k --tls--max 1.1 https://ghenriqu.42.fr

-- mariadb login
mariadb -u wp_user -p wordpress
SHOW TABLES
DESCRIBE wp_comments
SELECT comment_content FROM wp_comments

-- what is the PID?
docker exec nginx cat /proc/1/comm
